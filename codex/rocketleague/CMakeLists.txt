cmake_minimum_required(VERSION 3.20)
project(SM64RocketCar LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROCKETSIM "${CMAKE_CURRENT_SOURCE_DIR}/vendor/RocketSim")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_vendor.py"
  "${ROCKETSIM}" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/RocketSim.SHA256.json"
  RESULT_VARIABLE ROCKETSIM_VERIFY_STATUS)
if(NOT ROCKETSIM_VERIFY_STATUS EQUAL 0)
  message(FATAL_ERROR "Pinned RocketSim source verification failed")
endif()
add_subdirectory("${ROCKETSIM}" RocketSim)
# Preserve the audited vendor checkout. Build one generated translation unit
# with defined initial wheel state; no backend equations/constants are replaced.
set(ROCKET_VEHICLE_SOURCE "${ROCKETSIM}/src/Sim/btVehicleRL/btVehicleRL.cpp")
set(ROCKET_WHEEL_INIT "${CMAKE_CURRENT_SOURCE_DIR}/physics/wheel_init.inc")
set(ROCKET_KINEMATIC_SUSPENSION "${CMAKE_CURRENT_SOURCE_DIR}/physics/kinematic_suspension.inc")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROCKET_WHEEL_INIT}" "${ROCKET_KINEMATIC_SUSPENSION}" "${ROCKET_VEHICLE_SOURCE}")
file(READ "${ROCKET_VEHICLE_SOURCE}" ROCKET_VEHICLE_CODE)
file(READ "${ROCKET_WHEEL_INIT}" ROCKET_WHEEL_CODE)
file(READ "${ROCKET_KINEMATIC_SUSPENSION}" ROCKET_KINEMATIC_SUSPENSION_CODE)
set(ROCKET_WHEEL_ORIGINAL "m_wheelInfo.push_back(btWheelInfoRL(ci));")
string(FIND "${ROCKET_VEHICLE_CODE}" "${ROCKET_WHEEL_ORIGINAL}" ROCKET_WHEEL_OFFSET)
string(REGEX MATCHALL "m_wheelInfo[.]push_back[(]btWheelInfoRL[(]ci[)][)]" ROCKET_WHEEL_SITES "${ROCKET_VEHICLE_CODE}")
list(LENGTH ROCKET_WHEEL_SITES ROCKET_WHEEL_SITE_COUNT)
if(ROCKET_WHEEL_OFFSET LESS 0 OR NOT ROCKET_WHEEL_SITE_COUNT EQUAL 1)
  message(FATAL_ERROR "Pinned wheel initialization site changed")
endif()
string(REPLACE "${ROCKET_WHEEL_ORIGINAL}" "${ROCKET_WHEEL_CODE}" ROCKET_VEHICLE_CODE "${ROCKET_VEHICLE_CODE}")
# Adapt only the verified wheel-force site, using current triangle materials.
# Compatibility mode returns exactly 1; no pinned vendor file is modified.
set(ROCKET_GRIP_SITE "wheel.m_impulse = totalFrictionForce * frictionScale;")
string(REGEX MATCHALL "wheel[.]m_impulse = totalFrictionForce [*] frictionScale" ROCKET_GRIP_SITES "${ROCKET_VEHICLE_CODE}")
list(LENGTH ROCKET_GRIP_SITES ROCKET_GRIP_COUNT)
if(NOT ROCKET_GRIP_COUNT EQUAL 1)
  message(FATAL_ERROR "Pinned wheel impulse site changed")
endif()
string(REPLACE "${ROCKET_GRIP_SITE}" "wheel.m_impulse = totalFrictionForce * (frictionScale * rocket_host_wheel_grip(&wheel));" ROCKET_VEHICLE_CODE "${ROCKET_VEHICLE_CODE}")
string(PREPEND ROCKET_VEHICLE_CODE "extern \"C\" float rocket_host_wheel_grip(const void *wheel);\n")
set(ROCKET_SUSPENSION_ORIGINAL "float projVel = wheel.m_raycastInfo.m_contactNormalWS.dot(wheel.m_velAtContactPoint);")
string(FIND "${ROCKET_VEHICLE_CODE}" "${ROCKET_SUSPENSION_ORIGINAL}" ROCKET_SUSPENSION_OFFSET)
string(REGEX MATCHALL "float projVel = wheel[.]m_raycastInfo[.]m_contactNormalWS[.]dot[(]wheel[.]m_velAtContactPoint[)]" ROCKET_SUSPENSION_SITES "${ROCKET_VEHICLE_CODE}")
list(LENGTH ROCKET_SUSPENSION_SITES ROCKET_SUSPENSION_SITE_COUNT)
if(ROCKET_SUSPENSION_OFFSET LESS 0 OR NOT ROCKET_SUSPENSION_SITE_COUNT EQUAL 1)
  message(FATAL_ERROR "Pinned suspension-velocity site changed")
endif()
string(REPLACE "${ROCKET_SUSPENSION_ORIGINAL}" "${ROCKET_KINEMATIC_SUSPENSION_CODE}" ROCKET_VEHICLE_CODE "${ROCKET_VEHICLE_CODE}")
set(ROCKET_VEHICLE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/btVehicleRL_initialized.cpp")
file(CONFIGURE OUTPUT "${ROCKET_VEHICLE_GENERATED}" CONTENT "${ROCKET_VEHICLE_CODE}" @ONLY)
get_target_property(ROCKET_VENDOR_SOURCES RocketSim SOURCES)
list(FIND ROCKET_VENDOR_SOURCES "${ROCKET_VEHICLE_SOURCE}" ROCKET_VEHICLE_INDEX)
if(ROCKET_VEHICLE_INDEX LESS 0)
  message(FATAL_ERROR "Pinned vehicle translation unit missing from target")
endif()
list(REMOVE_ITEM ROCKET_VENDOR_SOURCES "${ROCKET_VEHICLE_SOURCE}")
set_property(TARGET RocketSim PROPERTY SOURCES "${ROCKET_VENDOR_SOURCES}" "${ROCKET_VEHICLE_GENERATED}")
set_source_files_properties("${ROCKET_VEHICLE_GENERATED}" TARGET_DIRECTORY RocketSim
  PROPERTIES INCLUDE_DIRECTORIES "${ROCKETSIM}/src/Sim/btVehicleRL")
add_library(rocket_physics STATIC physics/rocket_physics.cpp)
target_include_directories(rocket_physics PRIVATE "${ROCKETSIM}/src" "${ROCKETSIM}/libsrc/bullet3-3.24")
target_link_libraries(rocket_physics PUBLIC RocketSim)
foreach(ROCKET_TARGET RocketSim rocket_physics)
  if(MSVC)
    target_compile_options(${ROCKET_TARGET} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/physics/host_math_shim.h")
  else()
    target_compile_options(${ROCKET_TARGET} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/physics/host_math_shim.h")
  endif()
endforeach()
if(MSVC)
  target_compile_options(rocket_physics PRIVATE /W4 /fp:strict)
  target_compile_options(RocketSim PRIVATE /fp:strict)
else()
  target_compile_options(rocket_physics PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
  target_compile_options(RocketSim PRIVATE -fno-fast-math -ffp-contract=off)
endif()
enable_testing()
add_executable(rocket_water_test tests/test_water_physics.cpp)
target_link_libraries(rocket_water_test PRIVATE rocket_physics)
add_test(NAME rocket_water COMMAND rocket_water_test)
add_executable(rocket_environment_test tests/test_environment_physics.cpp)
target_link_libraries(rocket_environment_test PRIVATE rocket_physics)
add_test(NAME rocket_environment COMMAND rocket_environment_test)
add_executable(rocket_boost_mode_test tests/test_boost_physics.cpp)
target_link_libraries(rocket_boost_mode_test PRIVATE rocket_physics)
add_test(NAME rocket_boost_mode COMMAND rocket_boost_mode_test)
add_executable(rocket_physics_test tests/test_physics.cpp)
add_executable(rocket_whomp_physics_test tests/test_whomp_physics.cpp)
target_link_libraries(rocket_whomp_physics_test PRIVATE rocket_physics)
add_test(NAME rocket_whomp_physics COMMAND rocket_whomp_physics_test)
target_link_libraries(rocket_physics_test PRIVATE rocket_physics)
add_test(NAME rocket_physics COMMAND rocket_physics_test)
add_executable(rocket_platform_physics_test tests/test_platform_physics.cpp)
target_link_libraries(rocket_platform_physics_test PRIVATE rocket_physics)
add_test(NAME rocket_platform_physics COMMAND rocket_platform_physics_test)
add_executable(rocket_host_math_test tests/test_physics.cpp tests/poison_atan2f.c)
target_compile_definitions(rocket_host_math_test PRIVATE ROCKET_TEST_HOST_MATH=1)
target_link_libraries(rocket_host_math_test PRIVATE rocket_physics)
add_test(NAME rocket_host_math COMMAND rocket_host_math_test)

    ../../src/pc/network/packets/packet_read_write.c)
  target_include_directories(rocket_bindings_network_test PRIVATE ../.. ../../include
    ../../src ../../lib/lua/include ${ROCKET_NETWORK_SDL_INCLUDE})
  target_compile_definitions(rocket_bindings_network_test PRIVATE SDL_MAIN_HANDLED
    TARGET_PC VERSION_US NON_MATCHING AVOID_UB DISABLE_MODULE_LOG)
  target_link_libraries(rocket_bindings_network_test PRIVATE rocket_physics z)
  if(NOT MSVC)
    target_compile_options(rocket_bindings_network_test PRIVATE -fno-fast-math -ffp-contract=off)
  endif()
  add_test(NAME rocket_bindings_network COMMAND rocket_bindings_network_test)

add_executable(rocket_whomp_flip_physics_test tests/test_whomp_flip_physics.cpp ../../src/pc/character_net_codec.c)
